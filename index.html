import streamlit as st # ========== 页面配置（浏览器标签标题） ========== st.set_page_config( page_title="多囊饮食记录", page_icon="🍽", layout="wide" ) st.title("🍽多囊饮食&体重记录") # ========== 初始化会话存储 ========== if "food_list" not in st.session_state: st.session_state.food_list = [] if "exercise_list" not in st.session_state: st.session_state.exercise_list = [] if "weight_morning" not in st.session_state: st.session_state.weight_morning = "" if "weight_night" not in st.session_state: st.session_state.weight_night = "" # ========== 【1. 早晚体重录入区】 ========== st.subheader("⚖️ 今日体重") col_w1, col_w2 = st.columns(2) with col_w1: st.session_state.weight_morning = st.text_input("🌅 晨起体重(kg)", value=st.session_state.weight_morning) with col_w2: st.session_state.weight_night = st.text_input("🌙 晚间体重(kg)", value=st.session_state.weight_night) # ========== 【2. 食物录入区】 ========== st.subheader("🍱 添加饮食记录") col1, col2, col3, col4, col5, col6 = st.columns(6) with col1: meal = st.selectbox("餐次", ["🌞早餐","☀️午餐","🌙晚餐","🍬加餐"]) with col2: food_name = st.text_input("食物名称") with col3: kcal = st.number_input("热量kcal", min_value=0, value=0) with col4: protein = st.number_input("蛋白质g", min_value=0, value=0) with col5: carbs = st.number_input("碳水g", min_value=0, value=0) with col6: fat = st.number_input("脂肪g", min_value=0, value=0) if st.button("➕ 添加食物"): st.session_state.food_list.append({ "meal": meal, "name": food_name, "kcal": kcal, "protein": protein, "carbs": carbs, "fat": fat }) st.success("食物记录保存✅") # ========== 【3. 运动消耗录入区】 ========== st.subheader("🏃 运动消耗") ex_name = st.text_input("运动名称（可选）") ex_kcal = st.number_input("消耗热量(kcal)", min_value=0, value=0) if st.button("➕ 添加运动"): st.session_state.exercise_list.append({ "name": ex_name, "burn": ex_kcal }) st.success("运动记录保存✅") # ========== 【4. 汇总统计（一目了然） ========== st.divider() st.subheader("📊 今日汇总") # 计算总摄入 total_intake = sum(item["kcal"] for item in st.session_state.food_list) total_exercise = sum(item["burn"] for item in st.session_state.exercise_list) net_cal = total_intake - total_exercise col_sum1, col_sum2, col_sum3 = st.columns(3) with col_sum1: st.metric(label="🔥总摄入热量", value=f"{total_intake} kcal") with col_sum2: st.metric(label="💪运动总消耗", value=f"{total_exercise} kcal") with col_sum3: st.metric(label="📉净摄入热量", value=f"{net_cal} kcal") # 展示体重 col_wshow1, col_wshow2 = st.columns(2) with col_wshow1: st.write(f"🌅晨起体重：{st.session_state.weight_morning} kg") with col_wshow2: st.write(f"🌙晚间体重：{st.session_state.weight_night} kg") # ========== 【5. 明细列表】 ========== st.divider() st.subheader("📋饮食明细") for food in st.session_state.food_list: st.write(f"{food['meal']}｜{food['name']}｜{food['kcal']}kcal｜蛋白{food['protein']}g｜碳水{food['carbs']}g｜脂肪{food['fat']}g") st.subheader("🏃运动明细") for ex in st.session_state.exercise_list: st.write(f"{ex['name']}｜消耗 {ex['burn']} kcal") # 清空按钮 if st.button("🗑️清空今日全部记录"): st.session_state.food_list = [] st.session_state.exercise_list = [] st.session_state.weight_morning = "" st.session_state.weight_night = "" st.rerun()
